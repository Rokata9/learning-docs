Lesson 10.40 — Build: the dot product in Metal

Light theme · desktop · works from file://. Build lesson: no new idea; it joins Lessons 10.4–10.39.

Lessons 10.4–10.39 found dot(N, L), how far N goes along L, and showed it at every pixel as the dot map. Here you write both in Metal and run them on the iPhone.

Where this is used. The brightness of each pixel (Lessons 10.41–10.43), the lit glass of Lesson 10.47 and the highlight (Lessons 10.48–10.52) start from dot(N, L).

Section 2 · short code excerptThe iPhone shader

The shader is in chapter10/DotMap.metal; Swift calls it in chapter10/DotExplorer.swift. The highlighted lines are Lessons 10.38 and 10.39.

// chapter10/DotMap.metal — Lesson 10.3's N at one UV position (prepared)
static float3 circleNormalForDot(float2 uv, float2 center, float radius,
                                 float bevel, float step) { … }

[[ stitchable ]] half4 dotMap(
    float2 position, half4 color, float2 resolution, float2 center,
    float radius, float bevel, float step, float3 L, float scale
) {
    float2 uv = position * scale / resolution;
    float3 N = circleNormalForDot(uv, center, radius, bevel, step);
    float dotNL = N.x * L.x + N.y * L.y + N.z * L.z;
    float gray = dotNL * 0.5 + 0.5;
    return half4(gray, gray, gray, 1.0);
}
// chapter10/DotExplorer.swift — the call, each constant written as its value
ShaderLibrary.dotMap(
    .float2(pixelSize, pixelSize),   // resolution: the view is square
    .float2(0.50, 0.50),     // center C
    .float(0.25),            // radius
    .float(0.10),            // bevel
    .float(0.01),            // step
    .float3(0.48, -0.60, 0.64),   // L, the sliders' start
    .float(Float(displayScale))
)

float dotOfArrows(float3 a, float3 b) {
    float xMove = a.x * b.x;          // the x leg's move
    float yMove = a.y * b.y;          // the y leg's move
    float heightMove = a.z * b.z;     // the height leg's move
    float afterTwo = xMove + yMove;   // F₂
    return afterTwo + heightMove;     // F₃ = F: dot(a, b)
}

float buttonDotGray(float2 uv, float2 center, float2 outerHalf, float cornerRadius,
                    float bevel, float step, float3 L) {
    float3 N = buttonNormalForDot(uv, center, outerHalf, cornerRadius, bevel, step);
    float dotNL = dotOfArrows(N, L);
    return dotNL * 0.5 + 0.5;
}
